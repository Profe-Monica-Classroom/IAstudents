Definición del Modelo CNN

# Definición de la arquitectura
inputs <- layer_input(shape = c(32, 32, 1))

outputs <- inputs %>%
  
  layer_conv_2d(filters = 64, kernel_size = c(3,3), activation = "relu", padding = "same") %>%
  layer_conv_2d(filters = 64, kernel_size = c(3,3), activation = "relu", padding = "same") %>%
  layer_batch_normalization() %>%
  layer_max_pooling_2d(pool_size = c(2,2)) %>%
  layer_dropout(rate = 0.25) %>%
  
  layer_conv_2d(filters = 128, kernel_size = c(3,3), activation = "relu", padding = "same") %>%
  layer_conv_2d(filters = 128, kernel_size = c(3,3), activation = "relu", padding = "same") %>%
  layer_batch_normalization() %>%
  layer_max_pooling_2d(pool_size = c(2,2)) %>%
  layer_dropout(rate = 0.25) %>%
  
  layer_flatten() %>%
  layer_dense(units = 256, activation = "relu") %>%
  layer_dense(units = 64, activation = "relu") %>%
  layer_dropout(rate = 0.3) %>%
  layer_dense(units = 24, activation = "softmax")

model <- keras_model(inputs = inputs, outputs = outputs)

# Función para visualizar la arquitectura del modelo
plot_model_architecture <- function() {
  # Crear un gráfico simple que represente la arquitectura
  layers <- c(
    "Input (128x128x1)",
    "Conv2D (32 filtros)\n+ ReLU",
    "MaxPooling2D",
    "Conv2D (64 filtros)\n+ ReLU",
    "MaxPooling2D",
    "Conv2D (128 filtros)\n+ ReLU",
    "MaxPooling2D",
    "Flatten",
    "Dense (128)\n+ ReLU",
    "Dense (1)\n+ Sigmoid"
  )
  
  df <- data.frame(
    x = rep(1, length(layers)),
    y = seq(length(layers), 1),
    label = layers
  )
  
  ggplot(df, aes(x, y, label = label)) +
    geom_point(size = 10, color = "steelblue") +
    geom_text(color = "white", size = 3) +
    geom_segment(aes(x = x, xend = x, y = y-0.3, yend = y+0.3), 
                 arrow = arrow(length = unit(0.2, "cm")), 
                 color = "gray") +
    theme_void() +
    labs(title = "Arquitectura de la Red Neuronal") +
    theme(plot.title = element_text(hjust = 0.5))
}

# Visualizar algunas imágenes de ejemplo
plot_sample_images <- function(generator, n = 6) {
  batch <- generator_next(generator)
  images <- batch[[1]]
  labels <- batch[[2]]
  
  # Configurar área de gráfico
  par(mfrow = c(2, 3), mar = c(1, 1, 2, 1))
  
  for (i in 1:n) {
    # Extraer y normalizar la imagen (convertir a rango [0,1])
    img <- images[i,,,1]
    img_norm <- (img - min(img)) / (max(img) - min(img))
    
    # Crear plot con funciones base de R
    plot(NA, xlim = c(0, 1), ylim = c(0, 1), xaxt = 'n', yaxt = 'n', ann = FALSE)
    rasterImage(img_norm, 0, 0, 1, 1)
    title(main = ifelse(labels[i] == 1, classes[2], classes[1]), line = 0.5)
    box()
  }
}
